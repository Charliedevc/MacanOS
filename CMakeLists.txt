cmake_minimum_required(VERSION 3.21)
project(MacanOSMultiDisplay VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Network Gui Qml Quick)
add_library(displaycoordinator src/displaycoordinator.cpp src/displaycoordinator.h)
target_link_libraries(displaycoordinator PUBLIC Qt6::Core)
target_include_directories(displaycoordinator PUBLIC src)
add_executable(macanos-display-server src/server.cpp src/fixturemodel.cpp src/fixturemodel.h)
target_link_libraries(macanos-display-server PRIVATE displaycoordinator Qt6::Network)
qt_add_executable(macanos-display-client src/clientmain.cpp src/displayclient.cpp src/displayclient.h)
target_link_libraries(macanos-display-client PRIVATE Qt6::Core Qt6::Network Qt6::Gui Qt6::Qml Qt6::Quick)
file(GLOB QML_SOURCES CONFIGURE_DEPENDS qml/*.qml)
qt_add_resources(macanos-display-client display_qml PREFIX / BASE qml FILES ${QML_SOURCES})
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(tst_coordinator tests/tst_coordinator.cpp)
 target_link_libraries(tst_coordinator PRIVATE displaycoordinator Qt6::Test)
 add_test(NAME display_coordinator COMMAND tst_coordinator)
 qt_add_executable(tst_layout tests/tst_layout.cpp src/displayclient.cpp src/displayclient.h)
 target_include_directories(tst_layout PRIVATE src)
 target_link_libraries(tst_layout PRIVATE Qt6::Test Qt6::Network Qt6::Gui Qt6::Qml Qt6::Quick)
 qt_add_resources(tst_layout test_qml PREFIX / BASE qml FILES ${QML_SOURCES})
 add_test(NAME qml_layouts COMMAND tst_layout)
 set_tests_properties(qml_layouts PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
